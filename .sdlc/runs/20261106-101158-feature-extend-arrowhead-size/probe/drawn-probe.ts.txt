// Run-record probe (not project code): bounds vs every op point of the DRAWN shape,
// rotated about the element's real center (getElementAbsoluteCoords cx, cy).
import { arrayToMap } from "@excalidraw/common";
import { pointFrom, pointRotateRads } from "@excalidraw/math";
import { API } from "@excalidraw/excalidraw/tests/helpers/api";
import {
  getElementBounds,
  getElementAbsoluteCoords,
} from "@excalidraw/element/bounds";
import { ShapeCache } from "@excalidraw/element/shape";

import type { GlobalPoint, Radians, LocalPoint } from "@excalidraw/math";

const mk = (opts: Record<string, unknown>) =>
  API.createElement({
    type: "arrow",
    x: 0,
    y: 0,
    width: 100,
    height: 0,
    roughness: 0,
    points: [pointFrom<LocalPoint>(0, 0), pointFrom<LocalPoint>(100, 0)],
    ...opts,
  } as any) as any;

const check = (label: string, el: any) => {
  const map = arrayToMap([el]);
  const b = getElementBounds(el, map);
  const [, , , , cx, cy] = getElementAbsoluteCoords(el, map);
  const shapes = ShapeCache.generateElementShape(el, null) as any[];
  let worst = 0;
  for (const d of shapes.slice(1)) {
    for (const set of d.sets ?? []) {
      for (const op of set.ops ?? []) {
        const data: number[] = op.data;
        for (let i = 0; i + 1 < data.length; i += 2) {
          const [gx, gy] = pointRotateRads(
            pointFrom<GlobalPoint>(el.x + data[i], el.y + data[i + 1]),
            pointFrom<GlobalPoint>(cx, cy),
            el.angle as Radians,
          );
          worst = Math.max(worst, b[0] - gx, gx - b[2], b[1] - gy, gy - b[3]);
        }
      }
    }
  }
  console.log(`PROBE ${label}: heads=${shapes.length - 1} worstOutside=${worst.toFixed(3)}`);
};

it("probe", () => {
  for (const head of ["arrow", "triangle", "bar", "circle", "circle_outline", "diamond", "cardinality_many", "cardinality_one_or_many", "cardinality_zero_or_one", "cardinality_exactly_one", "cardinality_zero_or_many"]) {
    for (const angle of [0, Math.PI / 4, Math.PI / 2]) {
      check(`end ${head} angle=${angle.toFixed(2)} large`, mk({ angle, arrowheadSize: "large", endArrowhead: head }));
    }
  }
  check("start triangle angle=0.79 large", mk({ angle: Math.PI / 4, arrowheadSize: "large", startArrowhead: "triangle", endArrowhead: null }));
});
